# Paiza Mongolian IME — input profile: ZCode form code (形码).
#
# ZCode is not Unicode Mongolian. It is a form-code encoding in which each
# code point is one glyph component (tooth, belly, tail, …) and a written
# letter is a run of one to three of them. It borrows the 52 assigned code
# points U+1865..U+18AA of the Unicode Mongolian block, which belong to
# Sibe, Manchu and Ali Gali letters — so this text is only legible in a
# ZCode ("Z52") form-code font. See docs/input-profiles.md and
# docs/engine-contract.md.
#
# Selecting this profile needs two things installed that this file cannot
# provide: the engine executable (below) and a Z52 font.

id               zcode
display_name     ZCode (form code)

# Candidate-window font family, not a file: the user installs the font and
# this names it. A Unicode Mongolian font would render these code points as
# Sibe/Manchu letters, which is wrong text rather than a different look.
font             Z52 Hara Tig

# Per profile, and not optional: an entry learned as a run of glyph
# components is meaningless as Unicode Mongolian, and the other way round.
user_db          user-zcode.db

# --- the external engine (docs/engine-contract.md) -------------------------

# Executable name only, resolved inside the install directory — never a path,
# and PATH is never searched. Named *without* an extension: the host appends
# the platform's, so this one line serves all three platforms.
engine           zcode_stdio

# What the engine's HELLO must report. A mismatch disables this profile
# rather than letting a wrong binary produce text.
engine_protocol  paiza-engine

# Below the compiled lexicon by default; stated here because this profile has
# no lexicon of its own, so the engine is the only real source.
engine_priority  90

# Candidates outside this range are rejected. ZCode uses exactly these 52
# code points, so anything else means the engine is not what it claims.
engine_codepoints 1865-18AA

# `engine_transport` is deliberately omitted: it defaults per platform, and
# the engine sees plain stdin/stdout either way.

# What kind of answer to ask the engine for. The host passes this through
# unchanged, so the list of modes belongs to the engine, not to this build.
# zcode_stdio offers five (docs/engine-contract.md, and that engine's
# docs/stdio-protocol.md):
#
#   severe  letter shapes, the last letter in its final form
#   fuzzy   letter shapes, the last letter medial or final (still typing)
#   word    dictionary words, best score first
#   score   the same words, each with every metric behind its ranking
#   make    severe shapes and dictionary words merged
#   full    score, then the severe shapes it does not already contain
#
# `full` is the default, and the only one fit for interactive typing. The
# reasoning in order:
#
#   The scored modes are the only ones that say *why* a word ranks where it
#   does. Without that the host can only keep the order the engine sent,
#   which stops working the moment there is anything to merge in -- a user
#   word, a learned habit has to be placed *within* the engine's list.
#
#   But the dictionary holds 38,743 words, and people type words that are
#   not among them. `score` on its own then answers with nothing at all: a
#   blank candidate window and nothing to commit. (Measured: no candidates
#   for `surghaghuli`, for `bicig`, and for about a quarter of real word
#   prefixes.)
#
#   `full` appends the splicer's own reading of what was typed for exactly
#   that case, with the metrics zeroed, so the shapes sort below every real
#   word on the one scale and the window is never empty.
engine_mode      full

# Items in that mode are "text,frequency:...,score:..." rather than bare
# text, so the host takes the text up to the first comma and reads the score.
# Must match `engine_mode`: saying yes for a mode that answers plainly would
# cut candidates at a comma, and saying no for a scored mode would offer the
# metrics as if they were text (and every candidate would then fail the
# code-point check below, so the window would simply be empty).
engine_scored    yes

# The host may hand this engine the words the user has committed, so that
# the engine ranks them: it is the only place that decision can be made
# consistently with the engine's own scoring. Storage stays the host's --
# `user_db` above, one file per profile. Off by default for other profiles,
# because an engine need only implement the four requests the contract
# requires and one that never heard of user words should not be sent any.
engine_user_words yes

# How the engine weighs what has been typed before:
#
#   bonus = weight x count/(count + saturation) x 0.5^(days since / half_life)
#
# added to a score the engine normalises to at most 1, so `weight` is
# "how much a settled habit may be worth against a perfect dictionary
# match". 0 turns the habit off and still offers the user's own words.
engine_habit_weight      1.0

# Days after which an unused word's bonus halves. 10 and not 30 because of
# how far one use reaches: a single commit is worth 0.333, and a word the
# dictionary scored 0.938 needs only 0.062 to lead, so at 30 one stray
# choice would stay first for about 73 days. At 10 it leads for about 24 --
# long enough to be useful, short enough to be wrong harmlessly.
engine_habit_half_life   10.0

# At this many uses the habit is at half strength: smaller learns faster
# from the first use, larger wants to see a word several times. It never
# reaches full strength at any setting, so a word used a hundred times
# cannot bury one used five times for good.
engine_habit_saturation  2.0

# --- what this profile deliberately does not have -------------------------

# No `keymap`: a key mapping turns one key into fixed code points, and ZCode
# cannot be modelled that way — a component's identity depends on its
# position and its neighbours, which is what the engine computes.
#
# No `lexicon`: the compiled lexicon holds Unicode Mongolian words.

# GB/T appendix D.2 MVS insertion is a Unicode rule; it has no meaning here.
fixed_sequences  no

# The engine takes the raw Latin keys; there is no preedit to romanize.
romanize_query   no

# Consonant-skeleton input relies on the Unicode lexicon's vowels.
skeleton_input   no

# Nothing inline. The host now does source the preedit from the engine's best
# candidate on Windows (docs/input-profiles.md, increment 6c), so the
# candidate window's strip shows real form code -- but the *document* would
# render it in the application's own font, which TSF cannot set and which is
# not a Z52 font, so inline it would be Sibe and Manchu letters. The
# candidate window, which does use the profile's font, is the right place
# for it.
inline_default   hidden
